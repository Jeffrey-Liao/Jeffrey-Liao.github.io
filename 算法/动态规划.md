# 基础介绍
**通过组合子问题的解来求解原问题**

动态规划方法通常用来**求解最优化问题**。

动态规划方法是付出额外的内存来节省计算时间。

# 设计动态规划算法
1. 刻画一个最优解的结构
2. 递归地定义最优解的值
3. 计算最优的值
4. 利用计算信息构造最终解


# 实现方法
1. 带备忘的自顶向下法
	- 按照递归的书写方式，过程保存每个子问题的解
2. 自底向上法
	- 将子问题按照规模排序，由小到大的顺序进行求解

# 动态规划分类

## 一维动态规划
```cpp
// 示例：最大子数组和（Kadane算法）
// 状态定义：dp[i] = 以 nums[i] 结尾的最大子数组和
// 转移方程：dp[i] = max(nums[i], dp[i-1] + nums[i])
// 答案：max(dp[0..n-1])
int maxSubArray(vector<int>& nums) {
    int dp = nums[0];
    int ans = dp;
    for (int i = 1; i < nums.size(); i++) {
        dp = max(nums[i], dp + nums[i]);
        ans = max(ans, dp);
    }
    return ans;
}
```

## 二维路径 DP
```cpp
// 示例：不同路径（机器人从左上到右下）
// 状态定义：dp[i][j] = 到达 (i,j) 的路径数
// 转移方程：dp[i][j] = dp[i-1][j] + dp[i][j-1]
// 初始化：dp[0][j] = 1, dp[i][0] = 1

int uniquePaths(int m, int n) {
    vector<vector<int>> dp(m, vector<int>(n, 1));
    
    for (int i = 1; i < m; i++) {
        for (int j = 1; j < n; j++) {
            dp[i][j] = dp[i-1][j] + dp[i][j-1];
        }
    }
    return dp[m-1][n-1];
}
```

## 区间 DP
```cpp
// 示例：最长回文子序列
// 状态定义：dp[i][j] = 子串 s[i..j] 的最长回文子序列长度
// 转移方程：
//   if s[i] == s[j]: dp[i][j] = dp[i+1][j-1] + 2
//   else: dp[i][j] = max(dp[i+1][j], dp[i][j-1])
// 遍历顺序：按区间长度递增

int longestPalindromeSubseq(string s) {
    int n = s.length();
    vector<vector<int>> dp(n, vector<int>(n, 0));
    
    // 长度为1的子串
    for (int i = 0; i < n; i++) {
        dp[i][i] = 1;
    }
    
    // 按长度递增
    for (int len = 2; len <= n; len++) {
        for (int i = 0; i + len <= n; i++) {
            int j = i + len - 1;
            if (s[i] == s[j]) {
                dp[i][j] = dp[i+1][j-1] + 2;
            } else {
                dp[i][j] = max(dp[i+1][j], dp[i][j-1]);
            }
        }
    }
    return dp[0][n-1];
}
```

## 双序列 DP
```cpp
// 示例：最长公共子序列（LCS）
// 状态定义：dp[i][j] = text1[0..i-1] 和 text2[0..j-1] 的 LCS 长度
// 转移方程：
//   if text1[i-1] == text2[j-1]: dp[i][j] = dp[i-1][j-1] + 1
//   else: dp[i][j] = max(dp[i-1][j], dp[i][j-1])

int longestCommonSubsequence(string text1, string text2) {
    int m = text1.size(), n = text2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (text1[i-1] == text2[j-1]) {
                dp[i][j] = dp[i-1][j-1] + 1;
            } else {
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
            }
        }
    }
    return dp[m][n];
}
```

## 背包 DP
```cpp
// 0-1背包问题（每个物品选或不选）
// 状态定义：dp[j] = 容量为 j 时的最大价值
// 转移方程：dp[j] = max(dp[j], dp[j-w[i]] + v[i])
// 遍历顺序：物品在外，容量倒序

int knapsack01(int W, vector<int>& w, vector<int>& v) {
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < w.size(); i++) {
        for (int j = W; j >= w[i]; j--) {
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
        }
    }
    return dp[W];
}

// 完全背包（物品无限）
// 容量正序遍历
int knapsackComplete(int W, vector<int>& w, vector<int>& v) {
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < w.size(); i++) {
        for (int j = w[i]; j <= W; j++) {
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
        }
    }
    return dp[W];
}
```

## 树形 DP
```cpp
// 示例：二叉树的最大路径和
// 状态定义：对于每个节点，计算以该节点为根的路径最大值
// 转移：左子树贡献 + 右子树贡献 + 节点值

struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

int maxPathSum(TreeNode* root, int& maxSum) {
    if (!root) return 0;
    
    int left = max(0, maxPathSum(root->left, maxSum));
    int right = max(0, maxPathSum(root->right, maxSum));
    
    maxSum = max(maxSum, left + right + root->val);
    
    return max(left, right) + root->val;
}

int maxPathSum(TreeNode* root) {
    int maxSum = INT_MIN;
    maxPathSum(root, maxSum);
    return maxSum;
}
```

## 状态压缩 DP
```cpp
// 示例：旅行商问题（TSP）
// 状态定义：dp[mask][i] = 已访问集合 mask，当前在 i 的最短路径
// mask 的二进制位表示哪些城市已访问
// 转移：dp[mask][i] = min(dp[mask ^ (1<<i)][j] + dist[j][i])

int tsp(vector<vector<int>>& dist) {
    int n = dist.size();
    int fullMask = (1 << n) - 1;
    vector<vector<int>> dp(1 << n, vector<int>(n, INT_MAX));
    
    // 从城市0出发
    dp[1][0] = 0;
    
    for (int mask = 1; mask < (1 << n); mask++) {
        for (int i = 0; i < n; i++) {
            if (!(mask & (1 << i))) continue;
            if (dp[mask][i] == INT_MAX) continue;
            
            // 尝试访问未访问的城市
            for (int j = 0; j < n; j++) {
                if (mask & (1 << j)) continue;
                int newMask = mask | (1 << j);
                dp[newMask][j] = min(dp[newMask][j], dp[mask][i] + dist[i][j]);
            }
        }
    }
    
    // 返回起点
    int ans = INT_MAX;
    for (int i = 0; i < n; i++) {
        if (dp[fullMask][i] != INT_MAX) {
            ans = min(ans, dp[fullMask][i] + dist[i][0]);
        }
    }
    return ans;
}
```
